# randomForest tiene un limite de 53 niveles por factor.
collapse_levels <- function(x, top_n = 50) {
  freqs <- sort(table(x), decreasing = TRUE)
  keep  <- names(freqs)[seq_len(min(top_n, length(freqs)))]
  x <- as.character(x)
  x[!x %in% keep] <- "Otros"
  factor(x)
}
for (col in features_cat) {

    df_clean[[col]] <- collapse_levels(df_clean[[col]], top_n = 50)
  }
